// Copia esto en Firebase Console > Firestore > Rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Permitir que los usuarios lean/escriban en su propio documento y subcollecciones
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;

      // Fotos de progreso
      match /progressPhotos/{document=**} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }

      // Citas
      match /appointments/{document=**} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }
    }

    // Permitir leer products a cualquier usuario autenticado
    match /products/{document=**} {
      allow read: if request.auth != null;
      allow write: if false; // Solo admin en console
    }

    // Permitir leer routine_videos a cualquier usuario autenticado
    match /routine_videos/{document=**} {
      allow read: if request.auth != null;
      allow write: if false; // Solo administrador en console
    }

    // Permitir leer tip_videos a cualquier usuario autenticado
    match /tip_videos/{document=**} {
      allow read: if request.auth != null;
      allow write: if false;
    }

    // Foro/Comunidad: posts y comentarios
    match /Post/{postId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update: if request.auth != null; // incluye likes y commentsCount de otras usuarias
      allow delete: if request.auth != null && resource.data.autorId == request.auth.uid;

      match /comments/{commentId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null;
        allow update: if false;
        allow delete: if request.auth != null && resource.data.autorId == request.auth.uid;
      }
    }

    // Reportes de contenido del foro — solo se pueden crear, la revisión es manual
    // desde la consola de Firebase (nadie puede leer/editar/borrar reportes ajenos).
    match /reports/{reportId} {
      allow create: if request.auth != null;
      allow read, update, delete: if false;
    }

    // Registro de errores de la app (src/services/errorReporting.js) — mismo
    // patrón que reports: cualquiera logueada puede crear el suyo, la revisión
    // es manual desde la consola de Firebase, nadie puede leer/editar/borrar.
    match /error_logs/{errorId} {
      allow create: if request.auth != null;
      allow read, update, delete: if false;
    }

    // Directorio de peluquerías/tiendas — leído por my-beauty-calendar (Explorar) y
    // gestionado (creado/editado) por el panel de empresas (mycalendarbeauty-empresas).
    // Los documentos curados a mano (sembrados sin ownerId) quedan de solo-lectura:
    // no pasan el check de "update" y solo se editan desde la consola.
    match /peluquerias/{peluqueriaId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.ownerId == request.auth.uid;
      allow update: if request.auth != null && resource.data.ownerId == request.auth.uid;
      allow delete: if false;

      // Catálogo propio de la tienda (panel de empresas — "Mis productos").
      // Cualquiera logueada lo puede leer (para el matching del diagnóstico);
      // solo la dueña de la peluquería puede escribirlo.
      match /products/{productId} {
        allow read: if request.auth != null;
        allow write: if request.auth != null &&
          get(/databases/$(database)/documents/peluquerias/$(peluqueriaId)).data.ownerId == request.auth.uid;
      }

      // Servicios agendables (panel de empresas — "Mis servicios"). Cualquiera
      // logueada los puede leer (para elegir qué reservar); solo la dueña
      // de la peluquería los administra.
      match /servicios/{servicioId} {
        allow read: if request.auth != null;
        allow write: if request.auth != null &&
          get(/databases/$(database)/documents/peluquerias/$(peluqueriaId)).data.ownerId == request.auth.uid;
      }

      // Marcadores de horario ocupado (patrón de "slot lock" para reservar sin
      // choques). Cualquiera logueada los puede leer (para calcular horarios
      // libres); se crean solo para uno mismo (clienteUid propio, siempre
      // dentro de la misma transacción que crea la reserva); se borran al
      // cancelar, tanto por la clienta dueña del slot como por la dueña de
      // la tienda (así "Mi agenda" también puede liberar el horario).
      match /slots/{slotId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null && request.resource.data.clienteUid == request.auth.uid;
        allow delete: if request.auth != null && (
          resource.data.clienteUid == request.auth.uid ||
          get(/databases/$(database)/documents/peluquerias/$(peluqueriaId)).data.ownerId == request.auth.uid
        );
        allow update: if false;
      }
    }

    // Citas del cliente final — cada quien lee/actualiza (cancela) las suyas;
    // la dueña de la tienda también puede leer/actualizar (confirmar/cancelar)
    // las que le corresponden a su peluquería, para "Mi agenda" del panel de
    // empresas. El check de 'storeId' in resource.data es defensivo: las
    // reservas viejas (antes de este esquema) no tienen storeId, así que ese
    // lado del OR ni se evalúa para ellas — solo importa el check de clienteUid.
    match /reservas/{reservaId} {
      allow read, update: if request.auth != null && (
        resource.data.clienteUid == request.auth.uid ||
        ('storeId' in resource.data &&
         get(/databases/$(database)/documents/peluquerias/$(resource.data.storeId)).data.ownerId == request.auth.uid)
      );
      allow create: if request.auth != null && request.resource.data.clienteUid == request.auth.uid;
      allow delete: if false;
    }

    // Reseñas de peluquerías — cualquiera logueada puede leerlas (se muestran en el
    // perfil público) y crear la suya propia; no hay edición ni borrado hoy.
    match /resenias/{reseniaId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.clienteUid == request.auth.uid;
      allow update, delete: if false;
    }

    // Otras colecciones que puedas tener
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
